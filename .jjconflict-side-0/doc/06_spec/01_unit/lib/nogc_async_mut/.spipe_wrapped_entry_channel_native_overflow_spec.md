```simple
# @req REQ-SSPEC-LIB
step("send 1100 values, assert FIFO receive of every value")
val channel_id = rt_channel_new()
var sent = 1
while sent <= 1100:
    rt_channel_send(channel_id, sent)
    sent = sent + 1

var received = 1
var mismatches = 0
while received <= 1100:
    val value = rt_channel_try_recv(channel_id)
    if value != received:
        mismatches = mismatches + 1
    received = received + 1

rt_channel_close(channel_id)
expect(mismatches).to_equal(0)
```
