```simple
# @req REQ-SSPEC-LIB
step("round-trips all 256 byte values")
var b: [u8] = []
var i = 0
while i < 256:
    b.push((i % 256) as u8)
    i = i + 1
verify(file_write_bytes("/tmp/svllm_u8_full.bin", b))
val d = file_read_bytes("/tmp/svllm_u8_full.bin")
verify(d.len() == 256)
verify((d[0] as i64) == 0)
verify((d[200] as i64) == 200)
verify((d[255] as i64) == 255)
# oracle: last byte of the full 256-value pattern must round-trip exactly
expect((d[255] as i64) == 255).to_equal(true)
```
