DISK_CAP = 128 * 1024 * 1024

def stamp():
    return datetime.now(timezone.utc).isoformat()

def sha(path):
    h = hashlib.sha256()
    with open(path, "rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()

def value(v):
    if isinstance(v, bool): return "true" if v else "false"
    if v is None: return "null"
    if isinstance(v, (int, float)): return str(v)
    if isinstance(v, list): return "[\n" + ",\n".join(value(x) for x in v) + "\n]"
    if isinstance(v, dict): return "{\n" + ",\n".join(k + ": " + value(x) for k, x in v.items()) + "\n}"
    return json.dumps(str(v), ensure_ascii=False)

def save(name, record):
    (OUT / name).write_text(value(record) + "\n")

def git(*args):
    return subprocess.check_output(["git", *args], cwd=ROOT).decode().strip()

def inputs():
    paths = subprocess.check_output(["git", "ls-files", "-z", "src/runtime"], cwd=ROOT).decode().split("\0")
    excluded = {"src/runtime/miniaudio.h", "src/runtime/stb_image.h", "src/runtime/stb_truetype.h"}
    result = []
    for path in sorted(paths):
        if path and path.endswith((".c", ".h")) and "/vendor/" not in path and path not in excluded:
            result.append({"path": path, "sha256": sha(ROOT / path)})
    return result

def limits():
    resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
    resource.setrlimit(resource.RLIMIT_FSIZE, (64 * 1024 * 1024, 64 * 1024 * 1024))
    resource.setrlimit(resource.RLIMIT_AS, (2 * 1024 * 1024 * 1024, 2 * 1024 * 1024 * 1024))

initial_fixture_dirs = set(Path("/tmp").glob("simple-unix-scm-v1-*"))

def allocated():
    total = 0
    roots = [OUT] + list(set(Path("/tmp").glob("simple-unix-scm-v1-*")) - initial_fixture_dirs)
    for root in roots:
        for directory, _, files in os.walk(root):
            for path in [Path(directory)] + [Path(directory) / name for name in files]:
                try: total += path.stat(follow_symlinks=False).st_blocks * 512
                except FileNotFoundError: pass
    return total

def group_rss(group):
    total = 0
    for entry in Path("/proc").iterdir():
        if not entry.name.isdigit(): continue
        try:
            fields = (entry / "stat").read_text().rsplit(")", 1)[1].split()
            if int(fields[2]) != group: continue
            for row in (entry / "status").read_text().splitlines():
                if row.startswith("VmRSS:"): total += int(row.split()[1]) * 1024
        except (FileNotFoundError, ProcessLookupError, PermissionError): pass
    return total

def phase(name, command, timeout):
    environment = dict(os.environ)
    environment["TMPDIR"] = str(OUT / "tmp")
    started = stamp()
    with open(OUT / (name + ".stdout.txt"), "wb") as stdout, open(OUT / (name + ".stderr.txt"), "wb") as stderr:
        process = subprocess.Popen(command, cwd=ROOT, env=environment, stdout=stdout, stderr=stderr,
                                   start_new_session=True, preexec_fn=limits)
        record = {"phase": name, "started": started, "pid": process.pid, "processGroup": process.pid,
                  "argv": command, "cpuAffinity": "0-4", "tmpdir": environment["TMPDIR"]}
        save("live.sdn", record)
        print("START " + name + " pid=" + str(process.pid) + " pgid=" + str(process.pid), flush=True)
        begin = time.monotonic()
        peak_rss = 0
        peak_disk = 0
        reason = ""
        while process.poll() is None:
            peak_rss = max(peak_rss, group_rss(process.pid))
            peak_disk = max(peak_disk, allocated())
            if time.monotonic() - begin > timeout: reason = "timeout"
            if peak_rss > RSS_CAP: reason = "aggregate-rss-cap"
            if peak_disk > DISK_CAP: reason = "artifact-cap"
            if reason:
                try: os.killpg(process.pid, signal.SIGTERM)
                except ProcessLookupError: pass
                try: process.wait(timeout=15)
                except subprocess.TimeoutExpired:
                    try: os.killpg(process.pid, signal.SIGKILL)
                    except ProcessLookupError: pass
                    process.wait()
                break
            time.sleep(0.02)
        code = process.wait()
        # Cleanup only this admitted process group; no unrelated process IDs.
        try: os.killpg(process.pid, signal.SIGKILL)
        except ProcessLookupError: pass
        record.update({"finished": stamp(), "exitCode": code, "limitFailure": reason,
                       "elapsedSeconds": round(time.monotonic() - begin, 6),
                       "peakAggregateRssBytes": peak_rss, "peakAllocatedBytes": max(peak_disk, allocated())})
    record["stdoutSha256"] = sha(OUT / (name + ".stdout.txt"))
    record["stderrSha256"] = sha(OUT / (name + ".stderr.txt"))
    save(name + "-result.sdn", record)
    print("END " + name + " exit=" + str(code) + " limit=" + (reason or "none"), flush=True)
    return record
