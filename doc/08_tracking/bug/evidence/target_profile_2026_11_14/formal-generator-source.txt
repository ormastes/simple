extern fn rt_file_read_text(path: text) -> text?
extern fn rt_file_write_text(path: text, content: text) -> bool

struct LiveComponent:
    pos: i64
    name: text
    kind: text
    source: text

struct LiveGap:
    id: text
    row: text

fn extract_value(line: text, key: text) -> text:
    # find "key:" then take token up to ',' or '}'
    val marker = key + ":"
    val idx = line.index_of(marker)
    if idx < 0:
        return ""
    var rest = line.substring(idx + marker.len(), line.byte_len())
    rest = rest.trim()
    var out = ""
    var depth = 0
    for ch in rest.chars():
        if ch == "<":
            depth = depth + 1
        if ch == ">":
            depth = depth - 1
        if depth == 0 and (ch == "," or ch == "}"):
            break
        out = out + ch
    out.trim()

fn join_lines(lines: [text]) -> text:
    var out = ""
    for l in lines:
        out = out + l + "\n"
    out

# ---------------------------------------------------------------------------
# Canonical byte encoder (mirror of action_key.spl canon_* — same framing).
# Used ONLY to compute the golden-vector expected strings.
# ---------------------------------------------------------------------------

fn parse_live_schema(content: text) -> ([LiveComponent], [LiveGap], i64):
    var comps: [LiveComponent] = []
    var gaps: [LiveGap] = []
    var total: i64 = -1
    var in_comp = false
    var in_gap = false
    for line in content.split("\n"):
        val t = line.trim()
        if t.starts_with("inputs_total:"):
            total = t.substring(13, t.byte_len()).trim().to_i64()
            continue
        if t.starts_with("components:"):
            in_comp = true
            in_gap = false
            continue
        if t.starts_with("known_gaps:"):
            in_gap = true
            in_comp = false
            continue
        if t.starts_with("- {"):
            if in_comp:
                comps.push(LiveComponent(
                    pos: extract_value(t, "pos").to_i64(),
                    name: extract_value(t, "name"),
                    kind: extract_value(t, "kind"),
                    source: extract_value(t, "source")))
            if in_gap:
                gaps.push(LiveGap(id: extract_value(t, "id"), row: ""))
            continue
        if t == "" or t.starts_with("#"):
            continue
        in_comp = false
        in_gap = false
    (comps, gaps, total)

fn emit_live_key(comps: [LiveComponent], gaps: [LiveGap], total: i64, omit: text) -> text:
    var l: [text] = []
    l.push("/-")
    l.push("  CacheProtocol.Generated.LiveKey — GENERATED FILE. Do not edit.")
    l.push("  Generated by src/app/gen_cache_model/main.spl from")
    l.push("  src/compiler/80.driver/cache/schema/live_object_key.sdn.")
    l.push("  The LIVE native object/MIR cache key (producer identity, scope key,")
    l.push("  CompileOptionsHash, environment identity, DependencyEntry) as one")
    l.push("  framed component list, plus the counted inputs it does NOT capture.")
    l.push("  One `<name>_live_change_visible` per component, one `<id>_gap_invisible`")
    l.push("  per known gap; the gate checks |components| + |gaps| = inputs_total.")
    l.push("  Assumption stated, not proved: the Simple text encoders frame each")
    l.push("  component unambiguously (scope/producer identity length-framed by")
    l.push("  cache_key_frame; CompileOptionsHash by _coh_component) and digests are collision resistant.")
    l.push("-/")
    l.push("namespace CacheProtocol.Live")
    l.push("")
    l.push("/-- Live key components, schema order. -/")
    l.push("inductive Field where")
    for c in comps:
        l.push("  | {c.name}")
    l.push("  deriving DecidableEq, Repr")
    l.push("")
    l.push("def Field.name : Field → String")
    for c in comps:
        l.push("  | .{c.name} => \"{c.name}\"")
    l.push("")
    l.push("/-- scalar | ordered (order kept) | set (encoder sorts first). -/")
    l.push("def Field.kind : Field → String")
    for c in comps:
        l.push("  | .{c.name} => \"{c.kind}\"")
    l.push("")
    var fl = ""
    for c in comps:
        fl = if fl == "": ".{c.name}" else: fl + ", .{c.name}"
    l.push("def allFields : List Field := [{fl}]")
    l.push("")
    l.push("theorem mem_allFields (f : Field) : f ∈ allFields := by")
    l.push("  cases f <;> simp [allFields]")
    l.push("")
    l.push("/-- Inputs that can change an object but are not in the live key. -/")
    l.push("inductive Gap where")
    for g in gaps:
        l.push("  | {g.id}")
    l.push("  deriving DecidableEq, Repr")
    l.push("")
    var gl = ""
    for g in gaps:
        gl = if gl == "": ".{g.id}" else: gl + ", .{g.id}"
    l.push("def allGaps : List Gap := [{gl}]")
    l.push("")
    l.push("theorem mem_allGaps (g : Gap) : g ∈ allGaps := by")
    l.push("  cases g <;> simp [allGaps]")
    l.push("")
    l.push("def inputsTotal : Nat := {total}")
    l.push("")
    l.push("/-- §4.2 accounting: every enumerated input is a key component or a counted gap. -/")
    l.push("theorem inputs_accounted : allFields.length + allGaps.length = inputsTotal := by decide")
    l.push("")
    l.push("abbrev Val := List String")
    l.push("abbrev LiveKey := Field → Val")
    l.push("")
    l.push("/-- Framed canonical encoding: (component name, value) in schema order. -/")
    l.push("def enc (k : LiveKey) : List (String × Val) := allFields.map (fun f => (f.name, k f))")
    l.push("")
    l.push("theorem enc_inj {{k1 k2 : LiveKey}} (h : enc k1 = enc k2) : k1 = k2 := by")
    l.push("  funext f")
    l.push("  have hm := (List.map_inj_left.mp h) f (mem_allFields f)")
    l.push("  exact (Prod.mk.inj hm).2")
    for c in comps:
        if c.name == omit:
            l.push("")
            l.push("-- {c.name} DELIBERATELY OMITTED via --omit-field")
            continue
        l.push("")
        l.push("-- pos {c.pos} ({c.kind}) bound to {c.source}")
        l.push("theorem {c.name}_live_change_visible (k1 k2 : LiveKey)")
        l.push("    (h : k1 .{c.name} ≠ k2 .{c.name}) : enc k1 ≠ enc k2 :=")
        l.push("  fun e => h (congrFun (enc_inj e) .{c.name})")
    l.push("")
    l.push("/-- Everything a build may read: the live key plus the gap inputs. -/")
    l.push("structure Inputs where")
    l.push("  live : LiveKey")
    l.push("  gap  : Gap → Val")
    for g in gaps:
        l.push("")
        l.push("/-- Known gap: the live key does not see `{g.id}`. -/")
        l.push("theorem {g.id}_gap_invisible :")
        l.push("    ∃ i j : Inputs, enc i.live = enc j.live ∧ i.gap .{g.id} ≠ j.gap .{g.id} :=")
        l.push("  ⟨⟨fun _ => [], fun _ => []⟩, ⟨fun _ => [], fun _ => [\"changed\"]⟩, rfl, by simp⟩")
    l.push("")
    l.push("end CacheProtocol.Live")
    join_lines(l)

fn main() -> i64:
    val content = rt_file_read_text("live_object_key.sdn") ?? ""
    val (comps, gaps, total) = parse_live_schema(content)
    if comps.len() != 50 or gaps.len() != 0 or total != 50:
        return 1
    if not rt_file_write_text("LiveKey.lean", emit_live_key(comps, gaps, total, "")):
        return 2
    print "PASS actual live schema parser/emitter: 50 components"
    0
