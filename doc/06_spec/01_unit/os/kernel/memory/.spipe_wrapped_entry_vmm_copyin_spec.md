```simple
# @req REQ-SSPEC-UNIT
step("copies a byte range that crosses two mapped readable pages")
val pml4: u64 = 0x100000
val pdpt: u64 = 0x101000
val pd: u64 = 0x102000
val pt: u64 = 0x103000
val data0: u64 = 0x104000
val data1: u64 = 0x105000
_wire_test_user_pages(pml4, pdpt, pd, pt, 0x7000, data0, data1)
mmio_write64(data0 + 0xFF0, 0x4746454443424140)
mmio_write64(data0 + 0xFF8, 0x4F4E4D4C4B4A4948)
mmio_write64(data1, 0x6766656463626160)
mmio_write64(data1 + 8, 0x6F6E6D6C6B6A6968)
val area = VmArea(start: 0x7000, len: 8192, kind: VMA_ANON, flags: VMA_READ, backing: 0, backing_offset: 0)
var space = ProcessVmSpace(pml4: pml4, id: 6, vma_count: 0, areas: [])
space.vma_count = 1
space.areas.push(area)
val result = vmm_copyin_bytes_from_space(space, 0x7FF0, 32)
expect result.status.ok == true
expect result.bytes.len() == 32
expect result.bytes[0] == 0x40u8
expect result.bytes[15] == 0x4Fu8
expect result.bytes[16] == 0x60u8
expect result.bytes[31] == 0x6Fu8
```
