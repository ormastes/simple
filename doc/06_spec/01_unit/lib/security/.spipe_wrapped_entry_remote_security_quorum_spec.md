```simple
step("merges later expiry and repairs replicas on quorum read")
var newer = RemoteSecuritySessionStoreAdapter.replicated("redis", "security:newer")
newer.create_session("repair-session", "user-12", ["profile.read", "billing.write"], 9000)
var older = RemoteSecuritySessionStoreAdapter.replicated("redis", "security:older")
older.create_session("repair-session", "user-12", ["profile.read"], 4000)
var quorum = RemoteSecurityQuorumSessionStore.new([older, newer, older], 2, 2)

val winner = quorum.lookup_and_repair_session("repair-session")
val repaired = quorum.replicas[0].lookup_session("repair-session")

expect winner != nil
expect winner.unwrap().expires_at_ms == 9000
expect winner.unwrap().capabilities[1] == "billing.write"
expect repaired.unwrap().expires_at_ms == 9000
```
