```simple
# @req REQ-SSPEC-PERF
step("sub-parses each interpolation region exactly once, and no more")
# MECHANISM PIN, not a timing bound.
#
# A `lex_init_with_path` + `parse_expr` SUB-PARSE is the unit of real
# work in this phase. Each function generated by gen_body carries three
# string literals with two `{...}` regions each => exactly 6 regions per
# function that must ever be sub-parsed, and NOT one more.
#
# This is what regressed: `expand_interpolated_placeholder_call_args`
# re-ran the whole scan (and its sub-parses) over literals it could not
# act on, and the nodes it minted extended `expr_count()`, the bound of
# its own walk. Counting the sub-parses catches that directly -- a
# wall-clock budget alone cannot tell avoidable work from a slow host.
#
# None of these literals contains `_`, so the placeholder passes must
# do nothing at all.
si_stats_reset()
val m = parse_and_build_module_scoped(gen_body(10), "interp_subparse_pin.spl", false)
val literals = si_stat_literal_count()
val subparses = si_stat_subparse_count()
print "[perf] interp_subparse_pin: literals={literals} subparses={subparses}"
# 10 functions x 3 literals
expect(literals).to_equal(30)
# 10 functions x 3 literals x 2 regions, each sub-parsed exactly once
expect(subparses).to_equal(60)
```
